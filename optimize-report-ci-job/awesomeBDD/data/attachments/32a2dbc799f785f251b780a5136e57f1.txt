pm.test('session token was set', () => {
  const tok = pm.environment.get('sessionToken');
  pm.expect(tok).to.match(/^tok-/);
});
pm.test('user id was set', () => {
  const uid = pm.environment.get('userId');
  pm.expect(uid).to.match(/^usr-/);
});
pm.test('response ok', () => pm.response.to.have.status(200));
